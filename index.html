def fit_ols(x: pd.DataFrame, y: pd.Series):
    """Fit an ordinary least squares regression model using statsmodels,
    which reports R2, adjusted R2, F-statistic, probability of the
    F-statistic, coefficient estimates, and p-values for each
    coefficient in a single model summary."""
    x_with_const = sm.add_constant(x)
    model = sm.OLS(y, x_with_const).fit()
    return model


def backward_elimination(x: pd.DataFrame, y: pd.Series, significance: float = 0.05):
    """Optimize the regression model using backward stepwise elimination:
    repeatedly fit the model, identify the independent variable with the
    highest p-value, and remove it if that p-value exceeds the
    significance threshold. Repeat until all remaining variables are
    statistically significant."""
    remaining = list(x.columns)
    removal_order = []
    while True:
        model = fit_ols(x[remaining], y)
        p_values = model.pvalues.drop("const")
        max_p_value = p_values.max()
        if max_p_value > significance:
            variable_to_remove = p_values.idxmax()
            removal_order.append((variable_to_remove, round(max_p_value, 4)))
            remaining.remove(variable_to_remove)
        else:
            break
    return model, remaining, removal_order

def perform_pca(data_frame: pd.DataFrame, standardized_cols: list):
    """Perform PCA on the standardized continuous independent variables
    and return the fitted PCA object and the component scores."""
    pca = PCA(n_components=len(standardized_cols), random_state=42)
    component_scores = pca.fit_transform(data_frame[standardized_cols])
    return pca, component_scores


def determine_retained_components(pca: PCA) -> int:
    """Identify the number of principal components to retain using the
    Kaiser rule (retain components with an eigenvalue greater than 1)."""
    eigenvalues = pca.explained_variance_
    n_retained = int(np.sum(eigenvalues > 1))
    print("\n=== COMPONENT RETENTION (Kaiser rule: eigenvalue > 1) ===")
    for i, eigenvalue in enumerate(eigenvalues, start=1):
        flag = "RETAINED" if eigenvalue > 1 else ""
        print(f"  PC{i}: eigenvalue = {eigenvalue:.4f} {flag}")
    print(f"\nNumber of principal components retained: {n_retained}")
    return n_retained

def execute(self, context):
    aws_hook = AwsHook(self.aws_credentials_id)
    credentials = aws_hook.get_credentials()
    redshift = PostgresHook(postgres_conn_id=self.redshift_conn_id)

    if self.truncate_table:
        self.log.info(f'Clearing data from destination Redshift table {self.table}')
        redshift.run(f'TRUNCATE TABLE {self.table}')

    s3_path = f's3://{self.s3_bucket}/{self.s3_key}'
    self.log.info(f'Copying data from {s3_path} to Redshift table {self.table}')
    formatted_sql = StageToRedshiftOperator.copy_sql.format(
        self.table, s3_path, credentials.access_key,
        credentials.secret_key, self.region, self.json_path,
    )
    redshift.run(formatted_sql)
    self.log.info(f'Staging of {self.table} complete')

class LoadFactOperator(BaseOperator):
    """
    Loads a fact table in Redshift by running a provided SQL SELECT and
    inserting the results into the target table. Fact tables are large,
    so this operator only ever appends, it never truncates the
    destination table.
    """
    ui_color = '#F98866'

    def execute(self, context):
        redshift = PostgresHook(postgres_conn_id=self.redshift_conn_id)
        self.log.info(f'Loading fact table {self.table}')
        formatted_sql = LoadFactOperator.insert_sql.format(self.table, self.sql_query)
        redshift.run(formatted_sql)
        self.log.info(f'Load of fact table {self.table} complete')

class LoadDimensionOperator(BaseOperator):
    """
    Loads a dimension table in Redshift from a provided SQL SELECT.
    Defaults to the truncate-insert pattern (clear the table, then load
    fresh), but truncate_table can be set to False for append-only mode.
    """
    ui_color = '#80BD9E'

    def execute(self, context):
        redshift = PostgresHook(postgres_conn_id=self.redshift_conn_id)
        if self.truncate_table:
            self.log.info(f'Truncating dimension table {self.table}')
            redshift.run(f'TRUNCATE TABLE {self.table}')
        self.log.info(f'Loading dimension table {self.table}')
        formatted_sql = LoadDimensionOperator.insert_sql.format(self.table, self.sql_query)
        redshift.run(formatted_sql)
plugins/operators/data_quality.py
def execute(self, context):
    redshift = PostgresHook(postgres_conn_id=self.redshift_conn_id)
    if not self.dq_checks:
        raise ValueError('DataQualityOperator was given no checks to run')

    failing_checks = []
    for check in self.dq_checks:
        check_sql = check.get('check_sql')
        expected_result = check.get('expected_result')
        records = redshift.get_records(check_sql)
        if not records or not records[0]:
            failing_checks.append(f'Check returned no results. Query: {check_sql}')
            continue
        actual_result = records[0][0]
        if actual_result != expected_result:
            failing_checks.append(
                f'Check failed. Query: {check_sql} -- expected {expected_result}, got {actual_result}'
            )

    if failing_checks:
        raise ValueError('Data quality check(s) failed:\n' + '\n'.join(failing_checks))
    self.log.info(f'All {len(self.dq_checks)} data quality checks passed')

def calculate_debt_to_income_ratio(data_frame: pd.DataFrame) -> pd.DataFrame:
    """
    Calculate Debt to Income Ratio as Total Long-term Debt divided by
    Total Revenue. If Total Revenue is zero, the result is set to NaN
    to prevent division by zero and an invalid financial ratio.
    """
    ratio_results = data_frame[[
        "Business ID", "Total Long-term Debt", "Total Revenue"
    ]].copy()

    ratio_results["Debt to Income Ratio"] = np.where(
        ratio_results["Total Revenue"] == 0,
        np.nan,
        ratio_results["Total Long-term Debt"] / ratio_results["Total Revenue"]
    )
    return ratio_results
